        // 生成浮动方块（背景装饰）
        function createFloatingSquares() {
            const numSquares = 20;
            const container = document.querySelector('.container');
            for (let i = 0; i < numSquares; i++) {
                const square = document.createElement('div');
                square.classList.add('floating-square');
                square.style.left = `${Math.random() * 100}vw`;
                square.style.animationDelay = `${Math.random() * 10}s`;
                square.style.animationDuration = `${5 + Math.random() * 15}s`;
                square.style.width = `${10 + Math.random() * 30}px`;
                square.style.height = square.style.width;
                container.appendChild(square);
            }
        }
        createFloatingSquares();

        // 生成点阵文字
        function createDotText(text, id) {
            const container = document.getElementById(id);
            for (let i = 0; i < text.length; i++) {
                const char = text[i];
                if (char === ' ') {
                    const space = document.createElement('div');
                    space.style.width = '10px';
                    container.appendChild(space);
                    continue;
                }
                for (let j = 0; j < 5; j++) {
                    for (let k = 0; k < 5; k++) {
                        const dot = document.createElement('div');
                        dot.classList.add('dot');
                        // 模拟点阵（可根据需求调整显示逻辑）
                        if (Math.random() > 0.5) {
                            dot.style.backgroundColor = '#ffcce6'; // 隐藏的点
                        }
                        dot.style.animation = `fadeIn 0.5s ease ${(i * 5 + j) * 0.1 + k * 0.05}s forwards`;
                        container.appendChild(dot);
                    }
                    const br = document.createElement('br');
                    container.appendChild(br);
                }
            }
        }
        // 自定义问候文字
        const greetText1 = 'Hi 某某某';
        const greetText2 = '祝你';
        const greetText3 = '生日快乐';
        const greetText4 = 'happy birthday';
        createDotText(greetText1, 'greetText1');
        createDotText(greetText2, 'greetText2');
        createDotText(greetText3, 'greetText3');
        createDotText(greetText4, 'greetText4');

        // 页面切换函数
        function showSection(sectionId) {
            const sections = document.querySelectorAll('.section');
            sections.forEach(sec => {
                sec.classList.remove('active');
            });
            document.getElementById(sectionId).classList.add('active');
        }

        // 「开始」按钮点击事件：控制流程切换
        document.getElementById('startBtn').addEventListener('click', () => {
            showSection('section2');
            setTimeout(() => {
                showSection('section3');
                setTimeout(() => {
                    showSection('section4');
                    setTimeout(() => {
                        showSection('section5');
                        setTimeout(() => {
                            showSection('section6');
                            setTimeout(() => {
                                showSection('section7');
                                showWishes(); // 显示祝福语
                            }, 3000);
                        }, 2000);
                    }, 1000);
                }, 1000);
            }, 3000);
        });

        // 逐行显示祝福语
        function showWishes() {
            const wishes = [
                '愿你能精致到老，眼里长着太阳，笑里全是坦荡。',
                '阳光下永远像个孩子，风雨里像个大人，无畏亦无惧。',
                '你若不伤，岁月无恙。',
                '我不是在最好的时光遇见了你，而是遇见了你才开始了我这段最好的时光。',
                '承蒙时光不弃，感恩相遇。',
                '祝你：往下拉！'
            ];
            const wishesContainer = document.getElementById('wishesText');
            let index = 0;
            const interval = setInterval(() => {
                if (index < wishes.length) {
                    const p = document.createElement('p');
                    p.textContent = wishes[index];
                    p.style.opacity = '0';
                    p.style.transition = 'opacity 0.5s ease';
                    wishesContainer.appendChild(p);
                    setTimeout(() => {
                        p.style.opacity = '1';
                    }, 10);
                    index++;
                } else {
                    clearInterval(interval);
                }
            }, 1500);
        }

        // 「重放」按钮：回到欢迎页
        document.getElementById('repeatBtn').addEventListener('click', () => {
            showSection('section1');
        });

        // 「下载」按钮：保存为HTML文件（可选）
        document.getElementById('downloadBtn').addEventListener('click', () => {
            const html = document.documentElement.outerHTML;
            const blob = new Blob([html], { type: 'text/html' });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = '生日祝福.html';
            document.body.appendChild(a);
            a.click();
            document.body.removeChild(a);
            URL.revokeObjectURL(url);
        });
